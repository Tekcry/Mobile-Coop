```ts
// src/game/mapGate.ts - pure (no Babylon), the `world/menuGate.ts` pattern
export type MapStep = 'bake-live' | 'quality' | 'content' | 'bodies' | 'shaders' | 'probe' | 'bake-all' | 'audio' | 'warm' | 'peers' | 'ready';
/** S2's steps in order; S3 inserts 'audio' before 'warm', S5 appends 'peers'. Undo for one step (A1): drop it here. */
export const STEPS: readonly MapStep[] = ['bake-live', 'quality', 'content', 'bodies', 'shaders', 'probe', 'bake-all', 'warm'];
/** Never cut by the cap (A3): the live section's lamps (gameplay light). */
export const ALWAYS: readonly MapStep[] = ['bake-live'];
export const MAP_READY_CAP_MS = 20_000; // Michael's Q1 pick (A2), from the end of World.create
export const WARM_FRAMES = 5;
export interface MapGateInput { bakeLive: boolean; quality: boolean; content: boolean; bodies: boolean; shaders: boolean;
  probe: boolean; bakeAll: boolean; audio: boolean; peers: boolean; programs: number; warmFrames: number }
export interface MapGateState { step: MapStep; progress: number; text: string }
export function mapGateInput(): MapGateInput;
export function mapGateStep(g: MapGateInput, steps?: readonly MapStep[]): MapGateState;
/** `?readyCap=<s>` (5-600) else MAP_READY_CAP_MS. */
export function readyCapMs(search: string): number;
export interface MapGateProbe { read(g: MapGateInput): void; tick(): Promise<boolean>; closed(): boolean; describe(): string }
export interface MapGateResult { status: 'ready' | 'cap' | 'closed' | 'error'; capStep: MapStep | ''; ms: number }
export async function runMapGate(probe: MapGateProbe, report: (progress: number, text: string) => void, capMs: number,
  steps?: readonly MapStep[], now?: () => number): Promise<MapGateResult>;
// src/game/gameState.ts - NetAttachment (:177) gains (S5 implements them; unused until then):
  /** (map-ready S2) Co-op: this peer's release waits for the others (host: every start player's `mr`; client: `go`). */
  holdsStart?(): boolean;
  /** (map-ready S2) This peer's own gate finished (S5 sends `mr` from it). */
  onLocalReady?(info: { capHit: boolean; ms: number }): void;
```
Boot bar text per step: `bake-live` / `bake-all` "Building lights", `shaders` / `probe` "Compiling shaders", `bodies`
"Loading Wren", `quality` / `content` "Loading map...", `warm` "Final checks" (S5 adds "Waiting for partner").
`runMapGate`: never rejects; ticks until `ready`; past `capMs` it returns `cap` (with the step it was on) as soon as
every `ALWAYS` step is true (it keeps waiting for those, A3); `warm` counts rendered frames with every earlier step
true and `programs` unchanged (a change resets the count); a throw returns `error` (the boot screen never strands).
