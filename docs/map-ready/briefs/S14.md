## Interfaces
```ts
// src/core/compileLog.ts - pure (no Babylon, no DOM)
export type ShaderKind = 'body' | 'guard' | 'level' | 'lit' | 'post' | 'other';
/** From the effect's shader name and defines string: SMOOTHBODY -> body, GUARDFIG -> guard, LEVELSHADING or SURFACES ->
 *  level, BAKED_LAMPS (any other lit material) -> lit, a name other than 'pbr' / 'default' -> post, else other.
 *  Match whole define names (`#define SMOOTHBODY`), never substrings. */
export function shaderKindOf(name: string, defines: string): ShaderKind;
export interface CompileEntry {
  key: string;            // the cache key, cut to 200 chars
  kind: ShaderKind;
  defines: number;        // count of #define lines
  fragChars: number;      // final fragment source length when known, else -1
  startMs: number;        // first seen, ms from t0 (a key not ready at t0: 0, `early` true)
  readyMs: number;        // -1 while pending
  ms: number;             // readyMs - startMs, -1 while pending
  inFlight: number;       // entries pending when this one was first seen (itself excluded)
  early: boolean;
  failed: boolean;        // compile error or Babylon's fallback gave up
}
export interface CompileLogRecord {
  entries: CompileEntry[]; peakInFlight: number; pending: number; done: boolean;
  groups: Record<string, number>; // compile group -> its start, ms from t0 (S14's order, below)
  order: 'parallel' | 'serial';
}
export class CompileLog {
  readonly record: CompileLogRecord;
  constructor(t0: number, order: 'parallel' | 'serial');
  /** Keys compiled before t0 (already ready) are known, never logged. */
  known(key: string): void;
  seen(key: string, name: string, defines: string, now: number, early?: boolean): void; // a second call: ignored
  ready(key: string, now: number, fragChars: number, failed: boolean): void;            // unknown key: ignored
  group(name: string, now: number): void;    // first call per name kept
  finish(): void;                             // done = true; pending entries stay -1
  top(n: number): CompileEntry[];             // by ms, longest first; pending ones last
  /** e.g. "12 programs, peak 6 at once, longest body 31.2 s, sum 74.0 s over 33.5 s" */
  summary(now: number): string;
}

// src/game/compileOrder.ts - pure
export const GROUP_OWN = 1, GROUP_LEVEL = 2, GROUP_GUARDS = 4, GROUP_REST = 8, GROUP_ALL = 15;
export interface CompileDueInput { quality: boolean; content: boolean; bodies: boolean; started: number; settled: number }
/** The groups to start now (bits never already in `started`). parallel: LEVEL and GUARDS once quality && content,
 *  OWN once bodies && quality && content, REST once OWN, LEVEL and GUARDS have started; nothing waits for a group to
 *  settle. serial: S2's order: OWN once bodies && quality && content, then each next group only when the previous one
 *  has settled. */
export function compileGroupsDue(i: CompileDueInput, order: 'parallel' | 'serial'): number;
/** `?compileorder=serial` -> 'serial'; anything else -> 'parallel'. */
export function compileOrderOf(search: string): 'parallel' | 'serial';
```

## Steps
1. `src/core/compileLog.ts` and `tests/compileLog.test.ts` as above. `inFlight` and `peakInFlight` count pending
   entries; `summary` sums finished `ms` and gives the span from the first start to the last ready (the overlap factor
   the lead reads: sum / span). No allocation in `ready` / `seen` beyond the entry itself.
2. `src/game/compileOrder.ts` and `tests/compileOrder.test.ts` as above.
3. `gameState.ts`, the log (flag on or off, like S1's record; render side only, nothing in the sim reads it):
   - In `create`, beside S1's record (`window.__mapReady`): `this.compileLog = new CompileLog(<the S1 world-step
     time>, compileOrderOf(location.search))`; every key already in `engine._compiledEffects` whose effect `isReady()`
     goes to `known`; one not ready is `seen(..., early true)` and put on the pending list. Expose
     `window.__compileLog = log.record`.

     `spikeShaders` differs from the count the log last saw, walk `Object.keys(_compiledEffects)` once (load time
     only) and `seen` each new key with `effect.name` (a string, or the `fragment` / `fragmentElement` field of an
     object name) and `effect.defines`, adding it to a pending array (reused, never reallocated per frame); then for
     each pending effect: `isReady()` -> `ready(key, now, effect._fragmentSourceCode?.length ?? -1, false)`;
     `getCompilationError()` non-empty or disposed -> `ready(..., true)`; remove it (swap-remove). `finish()` when the
     pending list is empty and S1's after-window is done, or 150 s after t0; then drop every effect reference (S10
     keeps effects across matches: the log must not hold one). No work and no allocation per frame once done, and none
     on a frame with no new program and no pending entry.
